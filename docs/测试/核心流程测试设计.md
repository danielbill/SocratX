## 一、核心流程分析

### 1.1 核心流程概述

SocratX 的核心流程是 **用户消息 → AgentLoop → LLM → 工具调用 → 响应**：

```
用户消息
   ↓
AgentLoop.run() - 主入口
   ↓
1. 获取/创建会话 (SessionManager)
2. 添加用户消息到会话
3. 构建上下文 (ContextBuilder)
   ↓
AgentLoop._run_agent_loop() - 核心循环
   ↓
4. 调用 LLM (LiteLLMProvider.chat)
5. 检查是否有工具调用
   ├─ 有 → 执行工具 (ToolRegistry.execute)
   │      → 继续循环
   └─ 无 → 返回最终响应
   ↓
6. 保存会话
7. 返回 AgentResponse
```

### 1.2 核心组件

| 组件 | 文件 | 职责 |
|------|------|------|
| **AgentLoop** | `agent/loop.py` | 核心处理引擎，LLM 循环 |
| **LiteLLMProvider** | `providers/litellm_provider.py` | LLM 调用封装 |
| **SessionManager** | `agent/session.py` | 会话管理 |
| **MemoryStore** | `agent/memory.py` | 记忆存储 |
| **ToolRegistry** | `agent/tools/registry.py` | 工具注册与执行 |
| **ContextBuilder** | `agent/context.py` | 上下文构建 |

## 二、测试策略

### 2.1 测试原则

**隔离外部依赖 + 核心逻辑验证**

| 依赖 | Mock 策略 | 原因 |
|------|----------|------|
| LLM API | `MockLLMProvider` | 避免真实 API 调用，快速测试 |
| 文件系统 | `tmp_path` fixture | 隔离测试，自动清理 |
| 环境变量 | `monkeypatch` | 不污染全局环境 |

### 2.2 测试分类

```
核心流程测试
├── AgentLoop 测试（核心中的核心）
│   ├── 基础功能测试
│   ├── 工具调用测试
│   ├── 记忆系统测试
│   └── 边界条件测试
├── LiteLLMProvider 测试
│   ├── 初始化测试
│   ├── 聊天测试
│   └── 响应解析测试
└── 集成测试
    └── 端到端流程测试
```

```python
import pytest
from agent.loop import AgentLoop, AgentConfig, AgentResponse
from agent.session import SessionManager
from agent.memory import MemoryStore
from agent.tools.registry import ToolRegistry
from providers.litellm_provider import MockLLMProvider


class TestAgentLoopBasic:
    """AgentLoop 基础功能测试"""

    @pytest.fixture
    def components(self, tmp_path):
        """创建测试组件"""
        config = AgentConfig(model="mock-model")
        session_manager = SessionManager(base_dir=str(tmp_path))
        memory_store = MemoryStore(workspace=tmp_path)
        tool_registry = ToolRegistry()
        llm_provider = MockLLMProvider(response="Hello from mock!")

        return {
            "config": config,
            "session_manager": session_manager,
            "memory_store": memory_store,
            "tool_registry": tool_registry,
            "llm_provider": llm_provider,
        }

    @pytest.fixture
    def agent_loop(self, components):
        """创建 AgentLoop 实例"""
        loop = AgentLoop(
            config=components["config"],
            session_manager=components["session_manager"],
            memory_store=components["memory_store"],
            tool_registry=components["tool_registry"],
        )
        loop.set_llm_provider(components["llm_provider"])
        return loop

    @pytest.mark.asyncio
    async def test_run_simple_message(self, agent_loop):
        """测试简单消息处理"""
        response = await agent_loop.run(
            message="Hello",
            session_id="test-session",
            user_id="user-1",
        )

        assert isinstance(response, AgentResponse)
        assert response.content != ""
        assert len(response.tool_calls) == 0

    @pytest.mark.asyncio
    async def test_run_adds_to_session(self, agent_loop, components):
        """测试消息添加到会话"""
        await agent_loop.run(
            message="Test message",
            session_id="test-session",
        )

        session = components["session_manager"].get("test-session")
        assert session is not None
        assert len(session.messages) == 2  # 用户 + 助手

    @pytest.mark.asyncio
    async def test_run_without_llm_provider(self, agent_loop):
        """测试未设置 LLM 提供商"""
        agent_loop._llm_provider = None

        with pytest.raises(RuntimeError, match="LLM provider not set"):
            await agent_loop.run(message="Hello", session_id="test")
```

```python
    @pytest.mark.asyncio
    async def test_run_reaches_max_iterations(self, agent_loop, mocker):
        """测试达到最大迭代次数"""

        # Mock LLM 持续返回工具调用
```

```python
import pytest
from providers.litellm_provider import LiteLLMProvider, LLMResponse, MockLLMProvider


class TestLiteLLMProviderInit:
    """LiteLLMProvider 初始化测试"""

    def test_init_default(self):
        """测试默认初始化"""
        provider = LiteLLMProvider()

        assert provider.temperature == 0.7

    def test_init_with_api_key(self):

            api_key="sk-test-key",

        )
        
        assert provider.model == "anthropic/claude-3-5-sonnet"
        assert provider.api_key == "sk-test-key"
        assert provider.temperature == 0.5


class TestLiteLLMProviderChat:
    """LiteLLMProvider 聊天测试"""

    @pytest.mark.asyncio
    async def test_chat_basic(self, mocker):
        """测试基础聊天"""
        # Mock litellm.acompletion
        mock_response = mocker.MagicMock()
        mock_response.choices = [
            mocker.MagicMock(
                message=mocker.MagicMock(content="Hello!", tool_calls=None)
            )
        ]
        mock_response.usage = mocker.MagicMock(
            prompt_tokens=10, completion_tokens=20, total_tokens=30
        )
        mocker.patch("litellm.acompletion", return_value=mock_response)

        provider = LiteLLMProvider(model="gpt-4o")
        response = await provider.chat(
            messages=[{"role": "user", "content": "Hello"}]
        )

        assert response.content == "Hello!"
        assert response.model == "gpt-4o"
        assert response.usage["total_tokens"] == 30

    @pytest.mark.asyncio
    async def test_chat_with_tools(self, mocker):
        """测试带工具调用"""
        mock_response = mocker.MagicMock()
        mock_response.choices = [
            mocker.MagicMock(
                message=mocker.MagicMock(
                    content=None,
                    tool_calls=[
                        mocker.MagicMock(
                            id="call_1",
                            type="function",
                            function=mocker.MagicMock(
                                name="search", arguments='{"query": "test"}'
                            ),
                        )
                    ],
                )
            )
        ]
        mocker.patch("litellm.acompletion", return_value=mock_response)

        provider = LiteLLMProvider(model="gpt-4o")
        response = await provider.chat(
            messages=[{"role": "user", "content": "Search"}],
            tools=[{"type": "function", "function": {"name": "search"}}],
        )

        assert len(response.tool_calls) == 1
        assert response.tool_calls[0]["function"]["name"] == "search"
```

### 5.3 测试代码示例

```python
import pytest
from agent.loop import AgentLoop, AgentConfig
from agent.session import SessionManager
from agent.memory import MemoryStore
from agent.tools.registry import ToolRegistry, ToolBase
from providers.litellm_provider import MockLLMProvider


class TestTool(ToolBase):
    """测试工具"""
    
    async def execute(self, query: str = "default"):
        return f"Search result for: {query}"


class TestEndToEndFlow:
    """端到端流程测试"""

    @pytest.fixture
    def agent_loop(self, tmp_path, mocker):
        """创建完整的 AgentLoop"""
        config = AgentConfig(model="mock-model", max_iterations=5)
        session_manager = SessionManager(base_dir=str(tmp_path))
        memory_store = MemoryStore(workspace=tmp_path)
        tool_registry = ToolRegistry()
        
        # 注册测试工具
        tool_registry.register(TestTool())
        
        # Mock LLM 提供商
        llm_provider = mocker.MagicMock()
        
        loop = AgentLoop(
            config=config,
            session_manager=session_manager,
            memory_store=memory_store,
            tool_registry=tool_registry,
        )
        loop.set_llm_provider(llm_provider)
        
        return loop

    @pytest.mark.asyncio
    async def test_user_message_to_response(self, agent_loop, mocker):
        """测试用户消息到响应的完整流程"""
        # Mock LLM 响应
        mock_response = LLMResponse(content="Hello! How can I help?")
        agent_loop._llm_provider.chat = mocker.AsyncMock(return_value=mock_response)

        response = await agent_loop.run(
            message="Hello",
            session_id="test-session",
        )

        assert response.content != ""
        assert isinstance(response.content, str)

    @pytest.mark.asyncio
    async def test_conversation_with_history(self, agent_loop, mocker):
        """测试带历史对话的流程"""
        # 第一轮对话
        mock_response1 = LLMResponse(content="First response")
        agent_loop._llm_provider.chat = mocker.AsyncMock(return_value=mock_response1)
        
        await agent_loop.run(message="First message", session_id="test")

        # 第二轮对话
        mock_response2 = LLMResponse(content="Second response")
        agent_loop._llm_provider.chat = mocker.AsyncMock(return_value=mock_response2)
        
        response = await agent_loop.run(message="Second message", session_id="test")

        # 验证响应
        assert response.content == "Second response"
        
        # 验证会话中有 4 条消息（2 轮）
        session = agent_loop.session_manager.get("test")
        assert len(session.messages) == 4
```
